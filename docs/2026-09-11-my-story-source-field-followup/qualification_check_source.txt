def tally(path):
    cases = ET.parse(path).iter('testcase')
    return dict(Counter('failure' if c.find('failure') is not None else 'error'
        if c.find('error') is not None else 'xfail' if c.find('skipped') is not None
        and c.find('skipped').get('type') == 'pytest.xfail' else 'skip'
        if c.find('skipped') is not None else 'pass' for c in cases))

before = root / 'tmp/my_story_treatment_gap_full.xml'
after = root / 'tmp/source_field_final_full.xml'
b, c = failures(before), failures(after)
counts = tally(after)
assert counts == {'pass': 14411, 'skip': 183, 'failure': 51, 'xfail': 1}, counts
assert b.keys() == c.keys()
assert all(norm(b[k]) == norm(c[k]) for k in b), 'Changed inherited failure payload'
comparison = {'baseline': tally(before), 'final': counts,
              'new_failures': [], 'changed_normalized_failure_payloads': []}
(out / 'full_comparison.json').write_text(json.dumps(comparison, indent=2)+'\n', encoding='utf-8')
