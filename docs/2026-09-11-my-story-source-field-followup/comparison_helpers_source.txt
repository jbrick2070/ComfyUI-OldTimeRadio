import xml.etree.ElementTree as ET

         'tests/test_constrained_generate.py', 'tests/test_story_source_review.py']

def write(name, text):
    (OUT / name).write_text(text, encoding='utf-8')

def failures(path):
    tree = ET.parse(path)
    result = {}
    for case in tree.iter('testcase'):
        fail = case.find('failure')
        if fail is None:
            fail = case.find('error')
        if fail is not None:
            result[case.attrib['classname'] + '::' + case.attrib['name']] = fail.text or ''
    return result

def norm(text):
    text = re.sub(r'otr-source-field-(?:baseline|candidate)', 'PACK', text)
    text = re.sub(r'0x[0-9a-fA-F]+', 'ADDRESS', text)
    text = re.sub(r'pytest-[0-9]+', 'pytest-N', text)
    text = re.sub(r'pytest_[a-zA-Z0-9_]+[\\/]', 'pytest_TEMP/', text)
    # Pytest truncates this diagnostic repr at different characters because
    # the two worktree basenames have different lengths; assertions below
    # retain the actual finding lists and are still compared in full.
    text = re.sub(r'^py_files = .*$', 'py_files = <truncated fixture repr>', text, flags=re.M)
    # Pytest prints unequal mapping entries in process-dependent key order.
    # Sort only that explicit diagnostic block; retain each key/value pair.
    text = re.sub(r'(E\s+Differing items:\n)((?:E\s+\{[^\n]+\n)+)',
                  lambda match: match[1] + ''.join(sorted(match[2].splitlines(keepends=True))), text)
    return text
