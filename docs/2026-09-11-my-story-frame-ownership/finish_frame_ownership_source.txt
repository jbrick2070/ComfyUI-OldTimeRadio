from pathlib import Path
from collections import Counter
import ast
import hashlib
import json
import re
import shutil
import xml.etree.ElementTree as ET

r=Path.cwd();d=r/'docs/2026-09-11-my-story-frame-ownership'

def tally(p):
    return dict(Counter('failure' if c.find('failure') is not None else 'error'
        if c.find('error') is not None else 'xfail' if c.find('skipped') is not None
        and c.find('skipped').get('type')=='pytest.xfail' else 'skip'
        if c.find('skipped') is not None else 'pass' for c in ET.parse(p).iter('testcase')))

def norm(s):
    s=re.sub(r'otr-(?:source-(?:authority|feedback)|frame-ownership)-(?:baseline|candidate)','PACK',s)
    s=re.sub(r'0x[0-9a-fA-F]+','ADDRESS',s)
    s=re.sub(r'pytest-[0-9]+','pytest-N',s)
    s=re.sub(r'pytest_[a-zA-Z0-9_]+[\\/]','pytest_TEMP/',s)
    s=re.sub(r'^py_files = .*$', 'py_files = <truncated fixture repr>',s,flags=re.M)
    return re.sub(r'(E\s+Differing items:\n)((?:E\s+\{[^\n]+\n)+)',lambda m:m[1]+''.join(sorted(m[2].splitlines(keepends=True))),s)

def failures(p):
    result={}
    for c in ET.parse(p).iter('testcase'):
        f=c.find('failure')
        if f is None:f=c.find('error')
        if f is not None:result[norm(c.attrib['classname']+'::'+c.attrib['name'])]=norm(f.text or '')
    return result

def compare(a,b,name):
    old,new=failures(a),failures(b)
    result=dict(baseline=tally(a),candidate=tally(b),new_failures=sorted(new.keys()-old.keys()),
                fixed_guards=sorted(old.keys()-new.keys()),changed_normalized_failure_payloads=sorted(k for k in old.keys()&new.keys() if old[k]!=new[k]))
    (d/name).write_text(json.dumps(result,indent=2)+'\n',encoding='utf-8')
    print(name,json.dumps(result))
    assert not result['new_failures'] and not result['changed_normalized_failure_payloads'],result
    return result
