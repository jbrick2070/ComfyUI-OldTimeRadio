def tally(path):
    return dict(Counter('failure' if c.find('failure') is not None else 'error'
        if c.find('error') is not None else 'xfail' if c.find('skipped') is not None
        and c.find('skipped').get('type')=='pytest.xfail' else 'skip'
        if c.find('skipped') is not None else 'pass' for c in ET.parse(path).iter('testcase')))

def failures(path):
    result={}
    for c in ET.parse(path).iter('testcase'):
        f=c.find('failure')
        if f is None: f=c.find('error')
        if f is not None: result[c.attrib['classname']+'::'+c.attrib['name']]=f.text or ''
    return result

def norm(text):
    text=re.sub(r'otr-source-authority-(?:baseline|candidate)','PACK',text)
    text=re.sub(r'0x[0-9a-fA-F]+','ADDRESS',text)
    text=re.sub(r'pytest-[0-9]+','pytest-N',text)
    text=re.sub(r'pytest_[a-zA-Z0-9_]+[\\/]','pytest_TEMP/',text)
    text=re.sub(r'^py_files = .*$', 'py_files = <truncated fixture repr>',text,flags=re.M)
    text=re.sub(r'(E\s+Differing items:\n)((?:E\s+\{[^\n]+\n)+)',
                lambda m:m[1]+''.join(sorted(m[2].splitlines(keepends=True))),text)
    return text

def compare(before,after,name):
    b={norm(k):norm(v) for k,v in failures(before).items()}
    c={norm(k):norm(v) for k,v in failures(after).items()}
    record={'baseline':tally(before),'candidate':tally(after),
        'new_failures':sorted(c.keys()-b.keys()),'fixed_guards':sorted(b.keys()-c.keys()),
        'changed_normalized_failure_payloads':sorted(k for k in b.keys()&c.keys() if b[k]!=c[k])}
    (OUT/name).write_text(json.dumps(record,indent=2)+'\n',encoding='utf-8')
    return record

def evidence():
    checks=[]
    for relative in FILES:
        path=ROOT/relative;data=path.read_bytes()
        assert data and not data.startswith(b'\xef\xbb\xbf')
        text=data.decode('utf-8')
        if path.suffix=='.py': ast.parse(text,filename=relative,feature_version=(3,10))
        else: assert json.loads(json.dumps(json.loads(text)))==json.loads(text)
        checks.append({'path':relative,'sha256':hashlib.sha256(data).hexdigest(),
                       'utf8_no_bom':True,'python_310_parse':path.suffix=='.py'})
    canonical=ROOT/'workflows/otr_canonical.json'
    assert hashlib.sha256(canonical.read_bytes()).hexdigest()=='d586a286aaee4c039e410ae9a10014c5c7f4ab82d00eac0e9e1cc0564415057c'
    (OUT/'file_checks.json').write_text(json.dumps(checks,indent=2)+'\n',encoding='utf-8')
    comparisons={}
    comparisons['bible']=compare(ROOT/'tmp/source_authority_bible_baseline.xml',ROOT/'tmp/source_authority_bible_candidate.xml','bible_comparison.json')
    if (ROOT/'tmp/source_authority_full.xml').exists():
        try:
            comparisons['full']=compare(ROOT/'docs/2026-09-11-my-story-source-field-followup/tests/source_field_final_full.xml',ROOT/'tmp/source_authority_full.xml','full_comparison.json')
        except ET.ParseError:
            comparisons['full']='still running'
    else: comparisons['full']='still running'
    comparisons['focused']=tally(ROOT/'tmp/source_authority_focused_final.xml')
    comparisons['before_fix']=tally(ROOT/'tmp/source_authority_before_fix.xml')
    return comparisons
