"""CPU-only actual prompt routing with saved06 source and cloned act fixtures."""
from pathlib import Path
import copy
import hashlib
import json
import os
import sys

os.environ['OTR_TEST_MODE']='1'
os.environ['CUDA_VISIBLE_DEVICES']=''
root=Path(__file__).resolve().parents[1]
sys.path.insert(0,str(root))
from nodes import _otr_my_story as MS, _otr_story_input as SI, _otr_story_routing as RT

out=root/'docs/2026-09-11-my-story-source-authority'
path=root/'docs/2026-09-11-my-story-5080-qualification/pairlock_06_ledger.json'
data=json.loads(path.read_text(encoding='utf-8'))
raw=data['meta']['source_meta']['story_input']['fields']
story=data['meta']['my_story']
original=story['treatment']
act=next(row['returned_artifact'] for row in story['source_rewrites'] if row['pass_id']=='act_1')
records=[]
for count in (1,3,6):
    fixture=copy.deepcopy(original)
    fixture['acts']=[dict(copy.deepcopy(original['acts'][0]),n=i) for i in range(1,count+1)]
    treatment=MS.StoryTreatment.model_validate(fixture)
    bundle=SI.build_bundle(SI.capture_raw(**raw),SI.StoryRequest(act_count=str(count)))
    prev=None
    for index,plan in enumerate(treatment.acts):
        captured=[];journal=[]
        def slot(messages,**kwargs):
            assert 'source_rewrite_instruction' not in kwargs
            captured.append([dict(message) for message in messages])
            if messages[0]['content'].startswith('Check and rewrite'):
                payload=json.loads(messages[1]['content'])
                assert payload['source']=={k:raw.get(k,'') for k in SI.CREATIVE_FIELDS}
                return json.dumps(payload['draft'])
            return json.dumps(dict(copy.deepcopy(act),n=plan.n))
        result=MS._pass_act(slot,RT.resolve_story_pack('my_story'),bundle,treatment,
            plan,prev,treatment.acts[index-1] if index else None,must_speak=(),
            is_last=index==count-1,source_rewrite_receipts=journal)
        assert len(captured)==2 and len(journal)==1
        area,tail=captured[0][1]['content'].rsplit('\nACT SCOPE: ',1)
        scope=tail.split('\n\n',1)[0]
        expected=treatment.ending if index==count-1 and treatment.ending.strip() else plan.ending_state
        assert area.endswith('\n- where it should leave the story: '+expected)
        assert captured[1][0]['content'].count(scope)==1
        assert treatment.ending not in captured[1][0]['content']
        records.append({'fixture_act_count':count,'act_number':plan.n,
            'is_last':index==count-1,'must_speak':[],
            'saved06_local_ending_state':plan.ending_state,
            'saved06_global_ending':treatment.ending,'selected_explicit_target':expected,
            'static_scope':scope,'author_messages':captured[0],
            'correction_messages':captured[1],
            'journal':journal,'result_is_canned_fixture':True})
        prev=result
receipt={'kind':'CPU-only actual owner route, no model generation',
    'not_live_qualification':True,'source_ledger':str(path),
    'source_ledger_sha256':hashlib.sha256(path.read_bytes()).hexdigest(),
    'raw_source_fields':raw,'fixture_note':'Original06 act/plan cloned and renumbered to exercise1/3/6 routes; no new story quality evidence.',
    'routes':records}
(out/'offline_prompt_routes.json').write_text(json.dumps(receipt,ensure_ascii=False,indent=2)+'\n',encoding='utf-8')
print('10 act routes / 20 canned calls checked; exact saved source retained. No generation.')
