def sha(raw):
    return hashlib.sha256(raw).hexdigest()
def write_json(path, data):
    path.write_text(json.dumps(data, indent=2, ensure_ascii=False) + '\n', encoding='utf-8', newline='\n')
def git(where, *args):
    return subprocess.check_output(['git', *args], cwd=where)
def norm(text):
    return text.replace('\r\n', '\n')

assert results['baseline'] == results['candidate']
report['bible_strict_validator'] = {'baseline_exit': results['baseline']['exit_code'],

# Check active handoff links; dated archives deliberately keep their historical targets.
active = ['docs/GO_FORWARD_PLAN.md', 'session_handoff.md',
          'docs/2026-09-11-my-story-opus-handoff/README.md',
          'docs/2026-09-11-my-story-5080-qualification/pairlock_09_receipt.md']
checked_links = []
link_re = re.compile(r'\[[^\]\n]*\]\(([^)\n]+)\)')
for relative in active:
    path = root / relative
    for raw_target in link_re.findall(path.read_text(encoding='utf-8')):
        target = unquote(raw_target.strip('<>'))
        if re.match(r'\w+://', target):
            continue
        name, _, anchor = target.partition('#')
        resolved = (path.parent / name).resolve() if name else path
        assert resolved.exists(), (relative, target)
        if anchor:
            headings = re.findall(r'^#{1,6} (.+)$', resolved.read_text(encoding='utf-8'), re.M)
            slugs = [re.sub(r'[^\w\- ]', '', heading.lower()).replace(' ', '-') for heading in headings]
            assert anchor in slugs, (relative, target)
        checked_links.append({'file': relative, 'target': target})

# Only new links in broader living logs/README are this handoff's scope.
for relative in ['README.md', 'docs/HANDOFF_LOG.md', 'docs/PROD_BUG_LOG.md']:
    before = set(link_re.findall(git(root, 'show', 'HEAD:' + relative).decode('utf-8')))
    path = root / relative
    for target in set(link_re.findall(path.read_text(encoding='utf-8'))) - before:
        if re.match(r'\w+://', target):
            continue
        name = unquote(target.split('#')[0].strip('<>'))
        assert (path.parent / name).resolve().exists(), (relative, target)
        checked_links.append({'file': relative, 'target': target})
width = None
for line in (root / 'docs/GO_FORWARD_PLAN.md').read_text(encoding='utf-8').splitlines():
    if not line.startswith('|'):
        width = None
        continue
    count = len(re.split(r'(?<!\\)\|', line))
    if width is None:
        width = count
    assert count == width, line
report['markdown'] = {'active_and_new_links_checked': len(checked_links), 'broken_links': [],

def manifest(folder, candidates, filename):
    rows = []
    for path in sorted(candidates):
        if path.name == filename:
            continue
        raw = path.read_bytes()
        rows.append({'path': path.relative_to(root).as_posix(), 'bytes': len(raw), 'sha256': sha(raw)})
    write_json(folder / filename, {'scope': 'immutable evidence; excludes this manifest itself',
        'code_base': report['code_base'], 'file_count': len(rows), 'files': rows})
    return len(rows)
